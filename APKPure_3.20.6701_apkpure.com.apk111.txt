-- =========================================================-- WEEK 1 DATABASE ASSIGNMENT-- School Management Database-- =========================================================-- Create the databaseCREATE DATABASE school_management;-- Select the databaseUSE school_management;-- =========================================================-- 1. CREATE TABLES-- =========================================================-- Students tableCREATE TABLE students (    student_id INT AUTO_INCREMENT PRIMARY KEY,    first_name VARCHAR(50) NOT NULL,    last_name VARCHAR(50) NOT NULL,    gender VARCHAR(10),    date_of_birth DATE,    phone VARCHAR(20));-- Teachers tableCREATE TABLE teachers (    teacher_id INT AUTO_INCREMENT PRIMARY KEY,    first_name VARCHAR(50) NOT NULL,    last_name VARCHAR(50) NOT NULL,    subject VARCHAR(100),    phone VARCHAR(20));-- Classes tableCREATE TABLE classes (    class_id INT AUTO_INCREMENT PRIMARY KEY,    class_name VARCHAR(50) NOT NULL,    teacher_id INT,    FOREIGN KEY (teacher_id) REFERENCES teachers(teacher_id));-- Subjects tableCREATE TABLE subjects (    subject_id INT AUTO_INCREMENT PRIMARY KEY,    subject_name VARCHAR(100) NOT NULL);-- Enrollments tableCREATE TABLE enrollments (    enrollment_id INT AUTO_INCREMENT PRIMARY KEY,    student_id INT,    class_id INT,    enrollment_date DATE,    FOREIGN KEY (student_id) REFERENCES students(student_id),    FOREIGN KEY (class_id) REFERENCES classes(class_id));-- =========================================================-- 2. USEFUL QUERIES-- =========================================================-- Find all male studentsSELECT *FROM studentsWHERE gender = 'Male';-- Find all female studentsSELECT *FROM studentsWHERE gender = 'Female';-- Display students in alphabetical orderSELECT *FROM studentsORDER BY last_name ASC;-- Count the number of studentsSELECT COUNT(*) AS total_studentsFROM students;-- Count the number of teachersSELECT COUNT(*) AS total_teachersFROM teachers;-- Display classes together with their teachersSELECT    classes.class_name,    teachers.first_name,    teachers.last_nameFROM classesJOIN teachersON classes.teacher_id = teachers.teacher_id;-- Display students together with their classesSELECT    students.first_name,    students.last_name,    classes.class_name,    enrollments.enrollment_dateFROM enrollmentsJOIN studentsON enrollments.student_id = students.student_idJOIN classesON enrollments.class_id = classes.class_id;-- =========================================================-- END OF ASSIGNMENT-- =========================================================
